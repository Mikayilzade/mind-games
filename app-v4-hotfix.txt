  const __v41OriginalGetAdaptiveLevel = getAdaptiveLevel;
  getAdaptiveLevel = function(gameId) {
    const results = state.results.filter(result => result.gameId === gameId && !result.archived);

    if (gameId === 'sudoku') {
      if (!results.length) return 1;
      const last = results[results.length - 1];
      let level = Math.max(1, Math.min(7, Number(last.challengeSpec?.level) || 1));
      const hints = Number(last.stats?.hintsUsed || 0);
      const clean = last.mistakes === 0 && hints === 0 && last.accuracy >= 90;
      if (clean) level += 1;
      else if (last.accuracy < 70 || hints >= 2) level -= 1;
      return Math.max(1, Math.min(7, level));
    }

    if (gameId === 'sequence') {
      if (!results.length) return 1;
      const last = results[results.length - 1];
      const storedLevel = Number(last.challengeSpec?.level);
      const historicalLength = Number(last.challengeSpec?.sequence?.length || 4);
      let level = storedLevel || Math.max(1, Math.min(7, historicalLength - 3));
      if (last.mistakes === 0 && last.accuracy >= 95) level += 1;
      else if (last.accuracy < 70) level -= 1;
      return Math.max(1, Math.min(7, level));
    }

    return __v41OriginalGetAdaptiveLevel(gameId);
  };

      const level = getAdaptiveLevel('sequence');

      if (task.spec.sequence.length > targetLength) {
        task.spec.sequence = task.spec.sequence.slice(0, targetLength);
        changed = true;
      }
      if (task.spec.level !== level) {
        task.spec.level = level;
        changed = true;
      }
    }

  const __v41OriginalStartSudokuGame = startSudokuGame;
  startSudokuGame = function() {
    __v41OriginalStartSudokuGame();
    const stage = getGameStage();
    const board = stage?.querySelector('[data-sudoku-board]');
    if (!board) return;

    const syncCompletedDigits = () => {
      const counts = Array(10).fill(0);
      const conflicted = new Set();
      board.querySelectorAll('.sudoku-cell').forEach(cell => {
        const match = cell.getAttribute('aria-label')?.match(/число ([1-9])/);
        if (!match) return;
        const digit = Number(match[1]);
        counts[digit] += 1;
        if (cell.classList.contains('is-conflict')) conflicted.add(digit);
      });
      stage.querySelectorAll('[data-sudoku-number]').forEach(button => {
        const digit = Number(button.dataset.sudokuNumber);
        const complete = counts[digit] >= 9 && !conflicted.has(digit);
        button.disabled = complete;
        button.classList.toggle('is-complete', complete);
        button.setAttribute('aria-label', complete ? `Число ${digit}: все девять уже расставлены` : `Поставить число ${digit}`);
      });
    };

    const observer = new MutationObserver(syncCompletedDigits);
    observer.observe(board, { childList: true, subtree: true, attributes: true, attributeFilter: ['class', 'aria-label'] });
    syncCompletedDigits();

    const keyboardGuard = event => {
      if (!/^[1-9]$/.test(event.key)) return;
      const button = stage.querySelector(`[data-sudoku-number="${event.key}"]`);
      if (!button?.disabled) return;
      event.preventDefault();
      event.stopImmediatePropagation();
    };
    document.addEventListener('keydown', keyboardGuard, true);

    const previousCleanup = activeGame.cleanup;
    activeGame.cleanup = () => {
      observer.disconnect();
      document.removeEventListener('keydown', keyboardGuard, true);
      if (previousCleanup) previousCleanup();
    };
  };
